{{- /* A diagram too dense for the text column: spans the page width where no TOC sits beside it, and links to the full-size file. Usage: {{< diagram src="file.png" alt="..." caption="..." >}} The file is served under a fingerprinted name, so a redrawn diagram is never stale in a cache. */ -}} {{- $src := .Get "src" }} {{- $alt := .Get "alt" }} {{- $img := .Page.Resources.Get $src }} {{- if not $img }}{{ errorf "diagram: resource %q not found in %s" $src .Page.File.Path }}{{ end }} {{- $w := $img.Width }}{{ $h := $img.Height }} {{- $url := ($img | fingerprint "md5").RelPermalink }}
